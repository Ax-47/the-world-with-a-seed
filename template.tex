% ---------------------------------------------------------------
% Language and fonts (compile with XeLaTeX)
% ---------------------------------------------------------------
\usepackage{fontspec}
\usepackage{polyglossia}
\setdefaultlanguage{thai}
\setotherlanguage{english}

% Thai: Laksaman (TH Sarabun-style) reads well for long passages.
% English: TeX Gyre Pagella sits nicely next to it in weight and x-height.
\defaultfontfeatures{Ligatures=TeX}
\setmainfont{texgyrepagella}[Extension=.otf,
  UprightFont=*-regular, BoldFont=*-bold,
  ItalicFont=*-italic, BoldItalicFont=*-bolditalic]
\newfontfamily\englishfont{texgyrepagella}[Extension=.otf,
  UprightFont=*-regular, BoldFont=*-bold,
  ItalicFont=*-italic, BoldItalicFont=*-bolditalic]
\newfontfamily\thaifont[Script=Thai,Scale=1.2]{Laksaman}
\newfontfamily\headingfont[Script=Thai,Scale=1.15]{Kinnari}

% Thai line breaking
\XeTeXlinebreaklocale "th"
\XeTeXlinebreakskip = 0pt plus 1pt

% ---------------------------------------------------------------
% Page layout
% ---------------------------------------------------------------
\usepackage[a4paper,top=2.5cm,bottom=2.5cm,left=2.8cm,right=2.8cm,
  headheight=15pt]{geometry}
% Thai needs extra leading so stacked vowels and tone marks
% do not collide with the line above.
\newcommand{\thaistretch}{1.6}
\linespread{\thaistretch}

% Paragraph formatting: no indent, a bit of space between paragraphs
\usepackage[skip=0.6em plus 2pt]{parskip}

% ---------------------------------------------------------------
% Colours
% ---------------------------------------------------------------
\usepackage[dvipsnames]{xcolor}
\definecolor{seed}{HTML}{2E6B4F}      % deep green: headings, accents
\definecolor{seedlight}{HTML}{EEF5F0} % quote background
\definecolor{bark}{HTML}{8A5A2B}      % warm brown: notes
\definecolor{barklight}{HTML}{FBF5EC} % note background
\definecolor{inkgray}{HTML}{555555}

% ---------------------------------------------------------------
% Mathematics and graphics
% ---------------------------------------------------------------
\usepackage{graphicx}
\usepackage{amsmath,amssymb}

% ---------------------------------------------------------------
% Section headings
% ---------------------------------------------------------------
\usepackage{titlesec}
\titleformat{\section}
  {\headingfont\LARGE\bfseries\color{seed}}
  {\thesection}{0.8em}{}
  [{\color{seed}\titlerule[0.8pt]}]
\titleformat{\subsection}
  {\headingfont\Large\bfseries\color{seed}}
  {\thesubsection}{0.8em}{}
\titleformat{\subsubsection}
  {\headingfont\large\bfseries\color{inkgray}}
  {\thesubsubsection}{0.8em}{}
\titlespacing*{\section}{0pt}{1.5em}{1em}
\titlespacing*{\subsection}{0pt}{1.6em}{0.6em}
\titlespacing*{\subsubsection}{0pt}{1.2em}{0.4em}

% ---------------------------------------------------------------
% Lists
% ---------------------------------------------------------------
\usepackage{enumitem}
\setlist{itemsep=0.25em,topsep=0.4em}
\setlist[itemize,1]{label={\color{seed}\textbullet}}
\setlist[itemize,2]{label={\color{seed}\textendash}}

% ---------------------------------------------------------------

  \fancyfoot[C]{\small\color{inkgray}\thepage}}

% ---------------------------------------------------------------
% Boxes
% ---------------------------------------------------------------
\usepackage[most]{tcolorbox}

% Quotation from a book, with the source shown at the bottom right.
%   \begin{bookquote}{โลกของโซฟี่ หน้าที่ 22} ... \end{bookquote}
\newtcolorbox{bookquote}[1]{
  enhanced, breakable,
  colback=seedlight, colframe=seed,
  boxrule=0pt, leftrule=3pt, arc=0pt, outer arc=0pt,
  left=10pt, right=10pt, top=6pt, bottom=6pt,
  before skip=0.8em, after skip=0.8em,
  fontupper=\linespread{\thaistretch}\selectfont\itshape,
  parbox=false,
  after upper={\par\smallskip\hfill{\upshape\small\color{seed}--- จาก #1}},
}

% Personal commentary / observation.
%   \begin{note}[หัวข้อ] ... \end{note}
\newtcolorbox{note}[1][ข้อสังเกตของผม]{
  enhanced, breakable,
  colback=barklight, colframe=bark,
  boxrule=0.6pt, arc=2pt,
  left=10pt, right=10pt, top=6pt, bottom=6pt,
  before skip=0.8em, after skip=0.8em,
  parbox=false,
  fontupper=\linespread{\thaistretch}\selectfont,
  title={\headingfont\bfseries #1},
  coltitle=white, colbacktitle=bark,
  attach boxed title to top left={xshift=8pt,yshift=-2mm},
  boxed title style={arc=2pt,boxrule=0pt},
}
