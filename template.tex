% ---------------------------------------------------------------
% Language and fonts (compile with XeLaTeX)
% ---------------------------------------------------------------
\usepackage{fontspec}
\usepackage{polyglossia}
\setdefaultlanguage{thai}
\setotherlanguage{english}

% Thai: Sarabun (bundled in fonts/, SIL OFL). English: TeX Gyre Pagella.
\defaultfontfeatures{Ligatures=TeX}
\setmainfont{texgyrepagella}[Extension=.otf,
  UprightFont=*-regular, BoldFont=*-bold,
  ItalicFont=*-italic, BoldItalicFont=*-bolditalic]
\newfontfamily\englishfont{texgyrepagella}[Extension=.otf,
  UprightFont=*-regular, BoldFont=*-bold,
  ItalicFont=*-italic, BoldItalicFont=*-bolditalic]
\newfontfamily\thaifont{Sarabun}[Script=Thai, Scale=1.15,
  Path=fonts/, Extension=.ttf,
  UprightFont=*-Medium, BoldFont=*-Bold,
  ItalicFont=*-MediumItalic, BoldItalicFont=*-BoldItalic]

% Thai line breaking
\XeTeXlinebreaklocale "th"
\XeTeXlinebreakskip = 0pt plus 0.5pt
\setlength{\emergencystretch}{2em}
% Allow a line break where Thai runs straight into Latin letters or digits
% (e.g. "แต่Aristotle"), which ICU does not treat as a break point
\XeTeXinterchartokenstate=1
\newXeTeXintercharclass\thaicharclass
\newXeTeXintercharclass\latincharclass
\count255="0E01 \loop\ifnum\count255<"0E5C
  \XeTeXcharclass\count255=\thaicharclass \advance\count255 by 1 \repeat
\count255=`A \loop\ifnum\count255<`[
  \XeTeXcharclass\count255=\latincharclass \advance\count255 by 1 \repeat
\count255=`a \loop\ifnum\count255<`\{
  \XeTeXcharclass\count255=\latincharclass \advance\count255 by 1 \repeat
\count255=`0 \loop\ifnum\count255<`:
  \XeTeXcharclass\count255=\latincharclass \advance\count255 by 1 \repeat
\XeTeXinterchartoks\thaicharclass\latincharclass={\penalty50 }
\XeTeXinterchartoks\latincharclass\thaicharclass={\penalty50 }
% Break Thai only between words (ICU dictionary), never by hyphenating
\hyphenpenalty=10000
\exhyphenpenalty=10000

% ---------------------------------------------------------------
% Page layout
% ---------------------------------------------------------------
\usepackage[a4paper,top=2.5cm,bottom=2.5cm,left=2.8cm,right=2.8cm]{geometry}
% Thai needs extra leading so stacked vowels and tone marks
% do not collide with the line above.
\newcommand{\thaistretch}{1.6}
\linespread{\thaistretch}

% Paragraph formatting: indent every paragraph, including the first
\usepackage{indentfirst}
% polyglossia turns first-paragraph indent off again on every language
% switch; make its "off" switch do the same as "on"
\makeatletter
\let\nofrench@indent\french@indent
\makeatother
\setlength{\parindent}{1.5cm}
\setlength{\parskip}{0.3em}

% ---------------------------------------------------------------
% Mathematics and graphics
% ---------------------------------------------------------------
\usepackage{graphicx}
\usepackage{amsmath}
% Math in TeX Gyre Pagella Math, scaled to sit next to Sarabun
\usepackage{unicode-math}
\setmathfont{texgyrepagella-math.otf}[Scale=1.1]

% ---------------------------------------------------------------
% Lists
% ---------------------------------------------------------------
\usepackage{enumitem}
\setlist{itemsep=0.2em,topsep=0.3em}

% ---------------------------------------------------------------
% Quotes and notes
% ---------------------------------------------------------------
% Quotation from a book, with the source on its own line at the right.
%   \begin{bookquote}{โลกของโซฟี่ หน้าที่ 22} ... \end{bookquote}
\newenvironment{bookquote}[1]
  {\def\bookquotesource{#1}%
   \begin{list}{}{\setlength{\leftmargin}{2.5em}\setlength{\rightmargin}{2.5em}%
     \setlength{\listparindent}{1.5cm}\setlength{\parsep}{0.3em}}%
   \item\relax\linespread{\thaistretch}\selectfont\itshape}
  {\par\nopagebreak\hfill{\upshape --- จาก \bookquotesource}\end{list}}

% Personal commentary.
%   \begin{note}[หัวข้อ] ... \end{note}
\newenvironment{note}[1][ข้อสังเกตของผม]
  {\par\medskip\noindent\textbf{#1}\par}
  {\par\medskip}
